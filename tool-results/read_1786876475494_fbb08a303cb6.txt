   677→                      selectedOrg.actif 
   678→                        ? 'bg-green-100 text-green-700' 
   679→                        : 'bg-red-100 text-red-700'
   680→                    }`}>
   681→                      <span className={`w-1.5 h-1.5 rounded-full ${selectedOrg.actif ? 'bg-green-500' : 'bg-red-500'}`} />
