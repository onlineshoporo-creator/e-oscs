   149→   149→   149→// Composant Sparkline CSS
   150→   150→   150→function Sparkline({ data, color = '#F77F00', trend }: { data: number[]; color?: string; trend?: 'up' | 'down' | 'stable' }) {
   151→   151→   151→  const max = Math.max(...data)
   152→   152→   152→  const min = Math.min(...data)
   153→   153→   153→  const range = max - min || 1
   154→   154→   154→  
   155→   155→   155→  const points = data.map((value, index) => {
   156→   156→   156→    const x = (index / (data.length - 1)) * 100
   157→   157→   157→    const y = 100 - ((value - min) / range) * 100
   158→   158→   158→    return `${x},${y}`
   159→   159→   159→  }).join(' ')
   160→   160→   160→  
   161→   161→   161→  return (
   162→   162→   162→    <svg viewBox="0 0 100 40" className="w-full h-10 overflow-visible" preserveAspectRatio="none">
   163→   163→   163→      <defs>
   164→   164→   164→        <linearGradient id={`gradient-${color.replace('#', '')}`} x1="0%" y1="0%" x2="0%" y2="100%">
   165→   165→   165→          <stop offset="0%" stopColor={color} stopOpacity="0.3" />
   166→   166→   166→          <stop offset="100%" stopColor={color} stopOpacity="0" />
   167→   167→   167→        </linearGradient>
   168→   168→   168→      </defs>
   169→   169→   169→      {/* Area fill */}
   170→   170→   170→      <polygon
   171→   171→   171→        points={`0,100 ${points} 100,100`}
   172→   172→   172→        fill={`url(#gradient-${color.replace('#', '')})`}
   173→   173→   173→      />
   174→   174→   174→      {/* Line */}
   175→   175→   175→      <polyline
   176→   176→   176→        points={points}
   177→   177→   177→        fill="none"
   178→   178→   178→        stroke={color}
   179→   179→   179→        strokeWidth="2"
   180→   180→   180→        strokeLinecap="round"
   181→   181→   181→        strokeLinejoin="round"
   182→   182→   182→      />
   183→   183→   183→      {/* Last point dot */}
   184→   184→   184→      <circle cx="100" cy={100 - ((data[data.length - 1] - min) / range) * 100} r="3" fill={color} />
   185→   185→   185→    </svg>
   186→   186→   186→  )
   187→   187→   187→}
