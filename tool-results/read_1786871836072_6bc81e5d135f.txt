   159→export default function DemandesPage() {
   160→  const [requests, setRequests] = useState<SubscriptionRequest[]>([])
   161→  const [loading, setLoading] = useState(true)
   162→  const [filters, setFilters] = useState<Filters>({
   163→    statut: '',
   164→    region: '',
   165→    search: ''
   166→  })
   167→  const [pagination, setPagination] = useState<Pagination>({
   168→    page: 1,
   169→    limit: 10,
   170→    total: 0,
   171→    totalPages: 0
   172→  })
   173→  const [selectedRequest, setSelectedRequest] = useState<SubscriptionRequest | null>(null)
   174→  const [dialogOpen, setDialogOpen] = useState(false)
   175→  const [actionLoading, setActionLoading] = useState(false)
   176→  const [notesValue, setNotesValue] = useState('')
   177→
   178→  // Fetch requests
   179→  const fetchRequests = useCallback(async () => {
   180→    setLoading(true)
   181→    try {
   182→      const params = new URLSearchParams({
   183→        page: pagination.page.toString(),
   184→        limit: pagination.limit.toString(),
   185→        ...(filters.statut && { statut: filters.statut }),
   186→        ...(filters.region && filters.region !== 'Toutes les régions' && { region: filters.region }),
   187→        ...(filters.search && { search: filters.search })
   188→      })
   189→
   190→      const response = await fetch(`/api/admin/demandes?${params}`)
   191→      if (response.ok) {
   192→        const data = await response.json()
   193→        setRequests(data.data || [])
   194→        setPagination(prev => ({
   195→          ...prev,
   196→          total: data.total || 0,
   197→          totalPages: Math.ceil((data.total || 0) / prev.limit)
   198→        }))
   199→      }
   200→    } catch (error) {
   201→      console.error('Erreur lors de la récupération des demandes:', error)
   202→      toast.error('Erreur lors du chargement des demandes')
   203→    } finally {
   204→      setLoading(false)
   205→    }
   206→  }, [pagination.page, pagination.limit, filters])
   207→
   208→  useEffect(() => {
   209→    fetchRequests()
   210→  }, [fetchRequests])
   211→
   212→  // Actions sur une demande
   213→  const handleStatusUpdate = async (requestId: string, newStatus: RequestStatus, notes?: string) => {
   214→    setActionLoading(true)
   215→    try {
   216→      const response = await fetch(`/api/admin/demandes/${requestId}`, {
   217→        method: 'PATCH',
   218→        headers: { 'Content-Type': 'application/json' },
   219→        body: JSON.stringify({ statut: newStatus, notes_admin: notes || '' })
   220→      })
   221→
   222→      if (response.ok) {
   223→        toast.success(`Demande ${statusConfig[newStatus].label.toLowerCase()} avec succès`)
   224→        fetchRequests()
   225→        if (selectedRequest?.id === requestId) {
   226→          setSelectedRequest(prev => prev ? { ...prev, statut: newStatus, notes_admin: notes } : null)
   227→        }
   228→      } else {
   229→        throw new Error('Erreur lors de la mise à jour')
   230→      }
   231→    } catch (error) {
   232→      console.error(error)
   233→      toast.error('Erreur lors de la mise à jour du statut')
   234→    } finally {
   235→      setActionLoading(false)
   236→    }
   237→  }
   238→
   239→  // Approve request (create org + user + subscription)
   240→  const handleApproveRequest = async (requestId: string) => {
   241→    setActionLoading(true)
   242→    try {
   243→      const response = await fetch(`/api/admin/demandes/${requestId}/approve`, {
   244→        method: 'POST'
   245→      })
   246→
   247→      if (response.ok) {
   248→        toast.success('Organisation créée et abonnement activé avec succès')
   249→        setDialogOpen(false)
   250→        fetchRequests()
   251→      } else {
   252→        const error = await response.json()
   253→        throw new Error(error.message || 'Erreur lors de l\'approbation')
   254→      }
   255→    } catch (error) {
   256→      console.error(error)
   257→      toast.error(error instanceof Error ? error.message : 'Erreur lors de l\'approbation')
   258→    } finally {
   259→      setActionLoading(false)
   260→    }
   261→  }
   262→

   264→    return new Date(dateStr).toLocaleDateString('fr-FR', {
   265→      day: 'numeric',
   266→      month: 'short',
   267→      year: 'numeric',
   268→      hour: '2-digit',
   269→      minute: '2-digit'
   270→    })
   271→  }

   302→          </Card>
   303→        ))}

   330→                  <SelectItem key={key} value={key}>{config.label}</SelectItem>

   332→              </SelectContent>
   333→            </Select>
   334→
   335→            {/* Region filter */}
   336→            <Select value={filters.region} onValueChange={(value) => setFilters(prev => ({ ...prev, region: value }))}>

   340→              </SelectTrigger>

   342→                {regionsCI.map(region => (

   345→              </SelectContent>
   346→            </Select>
   347→          </div>
   348→        </CardContent>
   349→      </Card>

   354→          <div className="flex items-center justify-between">
   355→            <div>
   356→              <CardTitle className="text-lg font-semibold text-slate-900">
   357→                Liste des demandes
   358→              </CardTitle>
   359→              <CardDescription>
   360→                {pagination.total} demande(s) trouvée(s)
   361→              </CardDescription>
   362→            </div>
   363→          </div>
   364→        </CardHeader>
   365→        <CardContent>
   366→          {loading ? (
   367→            <div className="space-y-3">
   368→              {[...Array(5)].map((_, i) => (
   369→                <Skeleton key={i} className="h-20 w-full" />
   370→              ))}
   371→            </div>
   372→          ) : requests.length > 0 ? (
   373→            <>
   374→              {/* Desktop Table */}
   375→              <div className="hidden md:block overflow-x-auto">
   376→                <table className="w-full">
   377→                  <thead>
   378→                    <tr className="border-b border-slate-200">
   379→                      <th className="text-left py-3 px-4 text-xs font-semibold text-slate-500 uppercase tracking-wider">Date</th>
   380→                      <th className="text-left py-3 px-4 text-xs font-semibold text-slate-500 uppercase tracking-wider">Demandeur</th>
   381→                      <th className="text-left py-3 px-4 text-xs font-semibold text-slate-500 uppercase tracking-wider">Organisation</th>
   382→                      <th className="text-left py-3 px-4 text-xs font-semibold text-slate-500 uppercase tracking-wider">Type</th>
   383→                      <th className="text-left py-3 px-4 text-xs font-semibold text-slate-500 uppercase tracking-wider">Région</th>
   384→                      <th className="text-left py-3 px-4 text-xs font-semibold text-slate-500 uppercase tracking-wider">Statut</th>
   385→                      <th className="text-right py-3 px-4 text-xs font-semibold text-slate-500 uppercase tracking-wider">Actions</th>
   386→                    </tr>
   387→                  </thead>
   388→                  <tbody>
   389→                    {requests.map((request, index) => {
   390→                      const status = statusConfig[request.statut]
   391→                      return (
   392→                        <tr 
   393→                          key={request.id} 
   394→                          className={`border-b border-slate-100 hover:bg-slate-50 transition-colors ${
   395→                            index % 2 === 0 ? 'bg-white' : 'bg-slate-50/30'
   396→                          }`}
   397→                        >
   398→                          <td className="py-4 px-4">
   399→                            <div className="flex items-center gap-2 text-sm text-slate-600">
   400→                              <Calendar className="w-4 h-4 text-slate-400" />
   401→                              {formatDate(request.created_at)}
   402→                            </div>
   403→                          </td>
   404→                          <td className="py-4 px-4">
   405→                            <div>
   406→                              <p className="font-medium text-slate-900 flex items-center gap-2">
   407→                                <User className="w-4 h-4 text-slate-400" />
   408→                                {request.nom_complet}
   409→                              </p>
   410→                              <p className="text-xs text-slate-500 mt-0.5 flex items-center gap-1">
   411→                                <Mail className="w-3 h-3" />
   412→                                {request.email}
   413→                              </p>
   414→                            </div>
   415→                          </td>
   416→                          <td className="py-4 px-4">
   417→                            <p className="font-medium text-slate-900 flex items-center gap-2">
   418→                              <Building2 className="w-4 h-4 text-slate-400" />
   419→                              {request.nom_organisation}
   420→                            </p>
   421→                          </td>
   422→                          <td className="py-4 px-4">
   423→                            <Badge variant="outline" className="font-mono text-xs">
   424→                              {request.type_org}
   425→                            </Badge>
   426→                          </td>
   427→                          <td className="py-4 px-4">
   428→                            <span className="text-sm text-slate-600 flex items-center gap-1">
   429→                              <MapPin className="w-4 h-4 text-slate-400" />
   430→                              {request.region}
   431→                            </span>
   432→                          </td>
   433→                          <td className="py-4 px-4">
   434→                            <Badge variant="outline" className={`${status.className} text-xs`}>
   435→                              {status.icon && <status.icon className="w-3 h-3 mr-1" />}
   436→                              {status.label}
   437→                            </Badge>
   438→                          </td>
   439→                          <td className="py-4 px-4">
   440→                            <div className="flex items-center justify-end gap-1">
   441→                              <Dialog open={dialogOpen && selectedRequest?.id === request.id} onOpenChange={(open) => {
   442→                                setDialogOpen(open)
   443→                                if (open) {
   444→                                  setSelectedRequest(request)
   445→                                  setNotesValue(request.notes_admin || '')
   446→                                }
   447→                              }}>
   448→                                <DialogTrigger asChild>
   449→                                  <Button variant="ghost" size="icon" className="h-8 w-8">
   450→                                    <Eye className="w-4 h-4 text-slate-400" />
   451→                                  </Button>
   452→                                </DialogTrigger>
   453→                                <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   454→                                  <DialogHeader>
   455→                                    <DialogTitle>Détails de la demande</DialogTitle>
   456→                                    <DialogDescription>
   457→                                      Demande #{request.id.slice(0, 8)}...
   458→                                    </DialogDescription>
   459→                                  </DialogHeader>
   460→                                  
   461→                                  {selectedRequest && (
   462→                                    <div className="space-y-6 mt-4">
   463→                                      {/* Status badge */}
   464→                                      <div className="flex items-center justify-between p-4 rounded-lg bg-slate-50">

   471→                                      {/* Info grid */}
   472→                                      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   473→                                        <div className="space-y-3">
   474→                                          <div className="flex items-start gap-3">
   475→                                            <User className="w-5 h-5 text-slate-400 mt-0.5" />

   481→                                          <div className="flex items-start gap-3">

   778→              {(filters.search || filters.statut || filters.region) && (
