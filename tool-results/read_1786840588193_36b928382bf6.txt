   111→/**
   112→ * Composant principal ConnexionForm
   113→ */
   114→export function ConnexionForm() {
   115→  const router = useRouter()
   116→  const searchParams = useSearchParams()
   117→  const { success: showSuccess, error: showError } = useNotification()
   118→
   119→  // Onglet actif
   120→  const [activeTab, setActiveTab] = useState<AuthTab>('login')
   121→  
   122→  // États formulaire de connexion
   123→  const [loginEmail, setLoginEmail] = useState('')
   124→  const [loginPassword, setLoginPassword] = useState('')
   125→  const [isLoginPending, setIsLoginPending] = useState(false)
   126→  const [showPassword, setShowPassword] = useState(false)
   127→
   128→  // States for registration form
   129→  const [regData, setRegData] = useState({
   130→    nom_complet: '',
   131→    email: '',
   132→    telephone: '',
   133→    nom_organisation: '',
   134→    type_org: '' as OrgType | '',
   135→    region: '',
   136→    departement: '',
   137→    message: ''
   138→  })
   139→  const [isRegPending, setIsRegPending] = useState(false)
   140→  const [isRegSuccess, setIsRegSuccess] = useState(false)
   141→
   142→  // Récupérer l'URL de redirection
   143→  const redirectTo = searchParams.get('redirect') || '/app/dashboard'
   144→  
   145→  // Erreur depuis l'URL
   146→  const urlError = searchParams.get('error')
   147→
   148→  /**
   149→   * Soumission du formulaire de connexion
   150→   */
   151→  const handleLogin = async (e: React.FormEvent) => {
   152→    e.preventDefault()
   153→    
   154→    if (!loginEmail.trim()) {
   155→      showError('Champ requis', 'Veuillez entrer votre email.')
   156→      return
   157→    }
   158→    if (!loginPassword) {
   159→      showError('Champ requis', 'Veuillez entrer votre mot de passe.')
   160→      return
   161→    }
   162→
   163→    setIsLoginPending(true)
   164→    
   165→    try {
   166→      const response = await fetch('/api/auth', {
   167→        method: 'POST',
   168→        headers: { 'Content-Type': 'application/json' },
   169→        body: JSON.stringify({ action: 'signin', email: loginEmail, password: loginPassword })
   170→      })
   171→      
   172→      const result = await response.json()
   173→      
   174→      if (result.success) {
   175→        showSuccess('Connexion réussie !', 'Bienvenue sur e-OSCS. Redirection...', { duration: 3000 })
   176→        // Utiliser le redirectTo retourné par l'API ou celui par défaut
   177→        const redirectUrl = result.data?.redirectTo || redirectTo
   178→        setTimeout(() => {
   179→          router.push(redirectUrl)
   180→          router.refresh()
   181→        }, 500)
   182→      } else {
   183→        showError('Échec de la connexion', result.error || 'Email ou mot de passe incorrect.')
   184→      }
   185→    } catch (err) {
   186→      showError('Erreur réseau', 'Impossible de contacter le serveur. Veuillez réessayer.')
   187→    } finally {
   188→      setIsLoginPending(false)
   189→    }
   190→  }
   191→
   192→  /**
   193→   * Mise à jour champ inscription
   194→   */
   195→  const updateRegField = (field: string, value: string) => {
   196→    setRegData(prev => ({ ...prev, [field]: value }))
   197→  }
   198→
   199→  /**
   200→   * Validation formulaire inscription
   201→   */
   202→  const validateRegForm = (): string | null => {
   203→    if (!regData.nom_complet.trim()) return 'Le nom complet est requis.'
   204→    if (!regData.email.trim() || !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(regData.email)) {
   205→      return "L'email est invalide."
   206→    }
   207→    if (!regData.telephone.trim()) return 'Le numéro de téléphone est requis.'
   208→    if (!regData.nom_organisation.trim()) return "Le nom de l'organisation est requis."
   209→    if (!regData.type_org) return 'Veuillez sélectionner le type d\'organisation.'
   210→    if (!regData.region) return 'Veuillez sélectionner la région.'
   211→    return null
   212→  }
   213→
   214→  /**
   215→   * Soumission formulaire inscription
   216→   */
   217→  const handleRegister = async (e: React.FormEvent) => {
   218→    e.preventDefault()
   219→    
   220→    const validationError = validateRegForm()
   221→    if (validationError) {
   222→      showError('Formulaire incomplet', validationError)
   223→      return
   224→    }
   225→
   226→    setIsRegPending(true)
   227→    
   228→    try {
   229→      const response = await fetch('/api/auth', {
   230→        method: 'POST',
   231→        headers: { 'Content-Type': 'application/json' },
   232→        body: JSON.stringify({
   233→          action: 'request_access',
   234→          nom_complet: regData.nom_complet,
   235→          email: regData.email,
   236→          telephone: regData.telephone,
   237→          nom_organisation: regData.nom_organisation,
   238→          type_org: regData.type_org,
   239→          region: regData.region,
   240→          departement: regData.departement || undefined,
   241→          message: regData.message || undefined
   242→        })
   243→      })
   244→      
   245→      const result = await response.json()
   246→      
   247→      if (result.success) {
   248→        showSuccess('Demande envoyée !', 'Votre demande a été enregistrée avec succès.', { duration: 5000 })
   249→        setIsRegSuccess(true)
   250→      } else {
   251→        showError('Erreur', result.error || 'Erreur lors de la soumission.')
   252→      }
   253→    } catch (err) {
   254→      showError('Erreur réseau', 'Impossible de contacter le serveur.')
   255→    } finally {
   256→      setIsRegPending(false)
   257→    }
   258→  }
   259→
   260→  return (
   261→    <div className="w-full min-h-screen flex items-center justify-center bg-gradient-to-br from-gray-50 via-white to-orange-50/30 px-4 py-8 relative overflow-hidden">
   262→      {/* === Décorations de fond === */}
   263→      <div className="absolute inset-0 overflow-hidden pointer-events-none">
   264→        {/* Cercle orange */}
   265→        <div 
   266→          className="absolute -top-40 -right-40 w-80 h-80 rounded-full opacity-[0.04]"
   267→          style={{ backgroundColor: '#F77F00' }}
   268→        />
   269→        {/* Cercle vert */}
   270→        <div 
   271→          className="absolute -bottom-40 -left-40 w-96 h-96 rounded-full opacity-[0.04]"
   272→          style={{ backgroundColor: '#009E60' }}
   273→        />
   274→        {/* Grille subtile */}
   275→        <div 
   276→          className="absolute inset-0 opacity-[0.02]"
   277→          style={{
   278→            backgroundImage: `linear-gradient(#F77F00 1px, transparent 1px), linear-gradient(90deg, #F77F00 1px, transparent 1px)`,
   279→            backgroundSize: '60px 60px'
   280→          }}
   281→        />
   282→      </div>
   283→
   284→      {/* === Contenu principal === */}
   285→      <div className="relative z-10 w-full max-w-lg mx-auto">
   286→        
   287→        {/* Logo e-OSCS */}
   288→        <div className="flex justify-center mb-8">
   289→          <div className="relative group cursor-pointer" onClick={() => router.push('/')}>
   290→            <div className="absolute -inset-3 bg-gradient-to-br from-[#F77F00] via-transparent to-[#009E60] rounded-3xl opacity-20 group-hover:opacity-30 blur-xl transition-opacity duration-500" />
   291→            <div className="relative bg-white rounded-2xl p-4 shadow-xl border border-gray-100 hover:shadow-2xl transition-shadow duration-300">
   292→              <img
   293→                src="/logo-eoscs.png"
   294→                alt="e-OSCS - MCNSLP Côte d'Ivoire"
   295→                className="h-20 w-auto object-contain"
   296→              />
   297→            </div>
   298→          </div>
   299→        </div>
   300→
   301→        {/* Card principale */}
   302→        <div className="bg-white/95 backdrop-blur-sm rounded-2xl shadow-2xl overflow-hidden border border-gray-100">
   303→          
   304→          {/* Liseré tricolore CI */}
   305→          <div className="h-1.5 bg-gradient-to-r from-[#F77F00] via-white to-[#009E60]" />
   306→
   307→          <div className="p-8">
   308→            
   309→            {/* Titre principal */}
   310→            <div className="text-center mb-8">
   311→              <h1 className="text-2xl font-bold text-gray-900 mb-2">
   312→                Bienvenue sur <span className="bg-gradient-to-r from-[#F77F00] to-[#009E60] bg-clip-text text-transparent">e-OSCS</span>
   313→              </h1>
   314→              <p className="text-sm text-muted-foreground">
   315→                Plateforme de gestion du MCNSLP Côte d&apos;Ivoire
   316→              </p>
   317→            </div>
   318→
   319→            {/* === ONGLETS === */}
   320→            <div className="flex mb-8 bg-gray-100 rounded-xl p-1">
   321→              <button
   322→                onClick={() => setActiveTab('login')}
   323→                className={`flex-1 py-2.5 px-4 rounded-lg text-sm font-medium transition-all duration-200 ${
   324→                  activeTab === 'login'
   325→                    ? 'bg-white text-gray-900 shadow-sm'
   326→                    : 'text-gray-500 hover:text-gray-700'
   327→                }`}
   328→              >
   329→                <span className="flex items-center justify-center gap-2">
   330→                  <LockIcon className="w-4 h-4" />
   331→                  Connexion
   332→                </span>
   333→              </button>
   334→              <button
   335→                onClick={() => setActiveTab('register')}
   336→                className={`flex-1 py-2.5 px-4 rounded-lg text-sm font-medium transition-all duration-200 ${
   337→                  activeTab === 'register'
   338→                    ? 'bg-white text-gray-900 shadow-sm'
   339→                    : 'text-gray-500 hover:text-gray-700'
   340→                }`}
   341→              >
   342→                <span className="flex items-center justify-center gap-2">
   343→                  <UserIcon className="w-4 h-4" />
   344→                  Demander l&apos;accès
   345→                </span>
   346→              </button>
   347→            </div>
   348→
   349→            {/* === MESSAGE D'ERREUR URL === */}
   350→            {urlError && activeTab === 'login' && (
   351→              <div className="mb-6 p-3 rounded-lg bg-red-50 border border-red-200 text-red-700 text-sm text-center">
   352→                {urlError === 'oauth_failed' && 'La connexion OAuth a échoué.'}
   353→                {urlError === 'server_error' && 'Erreur serveur. Veuillez réessayer.'}
   354→                {!['oauth_failed', 'server_error'].includes(urlError) && urlError}
   355→              </div>
   356→            )}
   357→
   358→            {/* ======== ONGLET CONNEXION ======== */}
   359→            {activeTab === 'login' && (
   360→              <form onSubmit={handleLogin} className="space-y-5" noValidate>
   361→                
   362→                {/* Champ Email */}
   363→                <div className="space-y-2">
   364→                  <Label htmlFor="login-email" className="text-sm font-medium text-gray-700">
   365→                    Adresse email
   366→                  </Label>
   367→                  <div className="relative">
   368→                    <MailIcon className="absolute left-3 top-1/2 -translate-y-1/2 w-5 h-5 text-gray-400" />
   369→                    <Input
   370→                      id="login-email"
   371→                      type="email"
   372→                      placeholder="votre.email@example.com"
   373→                      value={loginEmail}
   374→                      onChange={(e) => setLoginEmail(e.target.value)}
   375→                      disabled={isLoginPending}
   376→                      autoComplete="email"
   377→                      required
   378→                      className="h-12 pl-11 bg-gray-50/50 border-gray-200 focus:border-[#F77F00] focus:ring-[#F77F00]/20"
   379→                    />
   380→                  </div>
   381→                </div>
   382→
   383→                {/* Champ Mot de passe */}
   384→                <div className="space-y-2">
   385→                  <div className="flex items-center justify-between">
   386→                    <Label htmlFor="login-password" className="text-sm font-medium text-gray-700">
   387→                      Mot de passe
   388→                    </Label>
   389→                    <button
   390→                      type="button"
   391→                      onClick={() => router.push('/mot-de-passe-oublie')}
   392→                      className="text-xs text-[#F77F00] hover:text-[#F77F00]/80 transition-colors"
   393→                    >
   394→                      Mot de passe oublié ?
   395→                    </button>
   396→                  </div>
   397→                  <div className="relative">
   398→                    <LockIcon className="absolute left-3 top-1/2 -translate-y-1/2 w-5 h-5 text-gray-400" />
   399→                    <Input
   400→                      id="login-password"
   401→                      type={showPassword ? 'text' : 'password'}
   402→                      placeholder="••••••••"
   403→                      value={loginPassword}
   404→                      onChange={(e) => setLoginPassword(e.target.value)}
   405→                      disabled={isLoginPending}
   406→                      autoComplete="current-password"
   407→                      required
   408→                      className="h-12 pl-11 pr-11 bg-gray-50/50 border-gray-200 focus:border-[#F77F00] focus:ring-[#F77F00]/20"
   409→                    />
   410→                    <button
   411→                      type="button"
   412→                      onClick={() => setShowPassword(!showPassword)}
   413→                      className="absolute right-3 top-1/2 -translate-y-1/2 text-gray-400 hover:text-gray-600 transition-colors"
   414→                    >
   415→                      {showPassword ? (
   416→                        <svg className="w-5 h-5" fill="none" viewBox="0 0 24 24" stroke="currentColor">
   417→                          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M13.875 18.825A10.05 10.05 0 0112 19c-4.478 0-8.268-2.943-9.543-7a9.97 9.97 0 011.563-3.029m5.858.908a3 3 0 114.243 4.243M9.878 9.878l4.242 4.242M9.88 9.88l-3.29-3.29m7.532 7.532l3.29 3.29M3 3l3.59 3.59m0 0A9.953 9.953 0 0112 5c4.478 0 8.268 2.943 9.542 7-1.274 4.057-5.064 7-9.542 7-4.477 0-8.268-2.943-9.542-7z" />
   418→                        </svg>
   419→                      ) : (
   420→                        <svg className="w-5 h-5" fill="none" viewBox="0 0 24 24" stroke="currentColor">
   421→                          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M15 12a3 3 0 11-6 0 3 3 0 016 0z" />
   422→                          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M2.458 12C3.732 7.943 7.523 5 12 5c4.478 0 8.268 2.943 9.542 7-1.274 4.057-5.064 7-9.542 7-4.477 0-8.268-2.943-9.542-7z" />
   423→                        </svg>
   424→                      )}
   425→                    </button>
   426→                  </div>
   427→                </div>
   428→
   429→                {/* Bouton Connexion */}
   430→                <Button
   431→                  type="submit"
   432→                  disabled={isLoginPending}
   433→                  className="w-full h-12 bg-gradient-to-r from-[#F77F00] to-[#E67E00] hover:from-[#E67E00] hover:to-[#F77F00] text-white font-semibold text-base shadow-lg hover:shadow-xl transition-all duration-300 rounded-xl"
   434→                >
   435→                  {isLoginPending ? (
   436→                    <span className="flex items-center gap-2">
   437→                      <svg className="animate-spin h-5 w-5" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24">
   438→                        <circle className="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" strokeWidth="4" />
   439→                        <path className="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z" />
   440→                      </svg>
   441→                      Connexion en cours...
   442→                    </span>
   443→                  ) : (
   444→                    'Se connecter'
   445→                  )}
   446→                </Button>
   447→
   448→                {/* Lien vers demande d'accès */}
   449→                <div className="text-center pt-2">
   450→                  <p className="text-sm text-gray-500">
   451→                    Pas encore de compte ?{' '}
   452→                    <button
   453→                      type="button"
   454→                      onClick={() => setActiveTab('register')}
   455→                      className="text-[#009E60] hover:text-[#009E60]/80 font-medium transition-colors"
   456→                    >
   457→                      Demander un accès
   458→                    </button>
   459→                  </p>
   460→                </div>
   461→              </form>
   462→            )}
   463→
   464→            {/* ======== ONGLET DEMANDE D'ACCÈS ======== */}
   465→            {activeTab === 'register' && (
   466→              <>
   467→                {isRegSuccess ? (
   468→                  /* État de succès */
   469→                  <div className="text-center space-y-5 py-8">
   470→                    <div className="w-20 h-20 mx-auto bg-green-100 rounded-full flex items-center justify-center">
   471→                      <svg className="w-10 h-10 text-[#009E60]" fill="none" viewBox="0 0 24 24" stroke="currentColor">
   472→                        <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M5 13l4 4L19 7" />
   473→                      </svg>
   474→                    </div>
   475→                    <div>
   476→                      <h3 className="text-xl font-bold text-gray-900 mb-2">Demande envoyée !</h3>
   477→                      <p className="text-gray-500 text-sm leading-relaxed">
   478→                        Votre demande a été enregistrée avec succès.<br />
   479→                        Nous la traiterons sous <strong className="text-gray-700">48 heures ouvrées</strong>.<br />
   480→                        Vous recevrez un email avec vos identifiants.
   481→                      </p>
   482→                    </div>
   483→                    <Button
   484→                      onClick={() => {
   485→                        setActiveTab('login')
   486→                        setIsRegSuccess(false)
