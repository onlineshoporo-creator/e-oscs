    57→// Types

    62→  email: string
    63→  telephone: string

    65→  fonction?: string
    66→  nom_organisation: string

   140→export default function DemandesPage() {
   141→  const [requests, setRequests] = useState<SubscriptionRequest[]>([])
   142→  const [loading, setLoading] = useState(true)
   143→  const [filters, setFilters] = useState<Filters>({
   144→    statut: '',
   145→    region: '',
   146→    search: ''
   147→  })
   148→  const [pagination, setPagination] = useState<Pagination>({
   149→    page: 1,
   150→    limit: 10,
   151→    total: 0,
   152→    totalPages: 0
   153→  })
   154→  const [selectedRequest, setSelectedRequest] = useState<SubscriptionRequest | null>(null)
   155→  const [dialogOpen, setDialogOpen] = useState(false)
   156→  const [actionLoading, setActionLoading] = useState(false)
   157→  const [notesValue, setNotesValue] = useState('')
   158→  const [stats, setStats] = useState<Record<string, number>>({})
   159→
   160→  // Fetch requests
   161→  const fetchRequests = useCallback(async () => {
   162→    setLoading(true)
   163→    try {
   164→      const params = new URLSearchParams({
   165→        page: pagination.page.toString(),
   166→        limit: pagination.limit.toString(),
   167→        ...(filters.statut && { statut: filters.statut }),
   168→        ...(filters.region && filters.region !== 'Toutes les régions' && { region: filters.region }),
   169→        ...(filters.search && { search: filters.search })
   170→      })
   171→
   172→      const response = await fetch(`/api/admin/demandes?${params}`)
   173→      if (response.ok) {
   174→        const data = await response.json()
   175→        setRequests(data.data || [])
   176→        setPagination(prev => ({
   177→          ...prev,
   178→          total: data.total || 0,
   179→          totalPages: Math.ceil((data.total || 0) / prev.limit)
   180→        }))
   181→        
   182→        // Calculer les stats localement
   183→        const allRequests = data.data || []
   184→        const statsObj: Record<string, number> = {}
   185→        Object.keys(statusConfig).forEach(key => {
   186→          statsObj[key] = allRequests.filter((r: SubscriptionRequest) => r.statut === key).length
   187→        })
   188→        statsObj['total'] = data.total || 0
   189→        setStats(statsObj)
   190→      } else {
   191→        toast.error('Erreur lors du chargement des demandes')
   192→      }
   193→    } catch (error) {
   194→      console.error('Erreur:', error)
   195→      toast.error('Erreur de connexion au serveur')
   196→    } finally {
   197→      setLoading(false)
   198→    }
   199→  }, [pagination.page, pagination.limit, filters])
   200→
   201→  useEffect(() => {
   202→    fetchRequests()
   203→  }, [fetchRequests])
   204→
   205→  // Actions sur une demande
   206→  const handleStatusUpdate = async (requestId: string, newStatus: RequestStatus, notes?: string) => {
   207→    setActionLoading(true)
   208→    try {
   209→      const response = await fetch(`/api/admin/demandes/${requestId}`, {
   210→        method: 'PATCH',
   211→        headers: { 'Content-Type': 'application/json' },
   212→        body: JSON.stringify({ statut: newStatus, notes_admin: notes || '' })
   213→      })
   214→
   215→      if (response.ok) {
   216→        toast.success(`Demande ${statusConfig[newStatus].label.toLowerCase()} avec succès`)
   217→        fetchRequests()
   218→        if (selectedRequest?.id === requestId) {
   219→          setSelectedRequest(prev => prev ? { ...prev, statut: newStatus, notes_admin: notes } : null)
   220→        }
   221→      } else {
   222→        throw new Error('Erreur lors de la mise à jour')
   223→      }
   224→    } catch (error) {
   225→      console.error(error)
   226→      toast.error('Erreur lors de la mise à jour du statut')
   227→    } finally {
   228→      setActionLoading(false)
   229→    }
   230→  }
   231→
   232→  // Approve request (create org + subscription)
   233→  const handleApproveRequest = async (requestId: string) => {
   234→    setActionLoading(true)
   235→    try {
   236→      const response = await fetch(`/api/admin/demandes/${requestId}`, {
   237→        method: 'POST'
   238→      })
   239→
   240→      if (response.ok) {
   241→        toast.success('Organisation créée et abonnement activé avec succès !')
   242→        setDialogOpen(false)
   243→        fetchRequests()
   244→      } else {
   245→        const error = await response.json()
   246→        throw new Error(error.message || "Erreur lors de l'approbation")
   247→      }
   248→    } catch (error) {
   249→      console.error(error)
   250→      toast.error(error instanceof Error ? error.message : "Erreur lors de l'approbation")
   251→    } finally {
   252→      setActionLoading(false)
   253→    }
   254→  }
   255→
   256→  const formatDate = (dateStr: string) => {
   257→    return new Date(dateStr).toLocaleDateString('fr-FR', {
   258→      day: 'numeric',
   259→      month: 'short',
   260→      year: 'numeric',
   261→      hour: '2-digit',
   262→      minute: '2-digit'
   263→    })
   264→  }
   265→
   266→  const getRelativeTime = (dateStr: string) => {
   267→    const now = new Date()
   268→    const date = new Date(dateStr)
   269→    const diffMs = now.getTime() - date.getTime()
   270→    const diffMins = Math.floor(diffMs / 60000)
   271→    const diffHours = Math.floor(diffMs / 3600000)
   272→    const diffDays = Math.floor(diffMs / 86400000)
   273→
   274→    if (diffMins < 60) return `Il y a ${diffMins} min`
   275→    if (diffHours < 24) return `Il y a ${diffHours}h`
   276→    return `Il y a ${diffDays}j`
   277→  }
   278→
   279→  return (
   280→    <div className="space-y-6">
   281→      {/* Page header */}
   282→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   283→        <div>
   284→          <h1 className="text-2xl lg:text-3xl font-bold text-slate-900">Demandes d&apos;abonnement</h1>
   285→          <p className="text-slate-500 mt-1">Gérez les demandes d&apos;accès à la plateforme e-OSCS</p>
   286→        </div>
   287→        <div className="flex items-center gap-3">
   288→          <Button variant="outline" onClick={fetchRequests} disabled={loading} className="gap-2">
   289→            <RefreshCw className={`w-4 h-4 ${loading ? 'animate-spin' : ''}`} />
   290→            Actualiser
   291→          </Button>
   292→          <Button variant="outline" className="gap-2" onClick={() => toast.info('Export CSV bientôt disponible')}>
   293→            <Download className="w-4 h-4" />
   294→            Exporter
   295→          </Button>
   296→        </div>
   297→      </div>
   298→
   299→      {/* Stats cards */}
   300→      <div className="grid grid-cols-2 md:grid-cols-5 gap-3 md:gap-4">
   301→        {/* Total */}
   302→        <Card className={`border-slate-200 ${!filters.statut ? 'ring-2 ring-orange-500' : ''}`}>
   303→          <CardContent 
   304→            className="p-4 cursor-pointer hover:bg-slate-50 transition-colors"
   305→            onClick={() => setFilters(prev => ({ ...prev, statut: '' }))}
   306→          >
   307→            <p className="text-xs text-slate-500 mb-1">Total</p>
   308→            <p className="text-2xl font-bold text-slate-900">{stats['total'] ?? '--'}</p>
   309→          </CardContent>
   310→        </Card>
   311→        
   312→        {Object.entries(statusConfig).map(([key, config]) => (
   313→          <Card key={key} className={`border-slate-200 ${filters.statut === key ? 'ring-2 ring-orange-500' : ''}`}>
   314→            <CardContent 
   315→              className="p-4 cursor-pointer hover:bg-slate-50 transition-colors"
   316→              onClick={() => setFilters(prev => ({ ...prev, statut: prev.statut === key ? '' : key }))}
   317→            >
   318→              <p className="text-xs text-slate-500 mb-1 flex items-center gap-1">
   319→                {config.icon && <config.icon className="w-3 h-3" />}
   320→                {config.label}s
   321→              </p>
   322→              <p className="text-2xl font-bold text-slate-900">{stats[key] ?? '--'}</p>
   323→            </CardContent>
   324→          </Card>
   325→        ))}
   326→      </div>
   327→
   328→      {/* Filters */}
   329→      <Card className="border-slate-200">
   330→        <CardContent className="p-4">
   331→          <div className="flex flex-col md:flex-row gap-4">
   332→            {/* Search */}
   333→            <div className="relative flex-1">
   334→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-slate-400" />
   335→              <Input
   336→                placeholder="Rechercher par nom, email ou organisation..."
   337→                value={filters.search}
   338→                onChange={(e) => setFilters(prev => ({ ...prev, search: e.target.value }))}
   339→                className="pl-10 bg-white"
   340→              />
   341→            </div>
   342→
   343→            {/* Status filter */}
   344→            <Select value={filters.statut} onValueChange={(value) => setFilters(prev => ({ ...prev, statut: value }))}>
   345→              <SelectTrigger className="w-full md:w-[180px]">
   346→                <Filter className="w-4 h-4 mr-2 text-slate-400" />
   347→                <SelectValue placeholder="Statut" />
   348→              </SelectTrigger>
   349→              <SelectContent>
   350→                <SelectItem value="">Tous les statuts</SelectItem>
   351→                {Object.entries(statusConfig).map(([key, config]) => (
   352→                  <SelectItem key={key} value={key}>{config.label}</SelectItem>
   353→                ))}
   354→              </SelectContent>
   355→            </Select>
   356→
   357→            {/* Region filter */}
   358→            <Select value={filters.region} onValueChange={(value) => setFilters(prev => ({ ...prev, region: value }))}>
   359→              <SelectTrigger className="w-full md:w-[200px]">
   360→                <MapPin className="w-4 h-4 mr-2 text-slate-400" />
   361→                <SelectValue placeholder="Région" />
   362→              </SelectTrigger>
   363→              <SelectContent>
   364→                {regionsCI.map(region => (
   365→                  <SelectItem key={region} value={region}>{region}</SelectItem>
   366→                ))}
   367→              </SelectContent>
   368→            </Select>
   369→          </div>
   370→        </CardContent>
   371→      </Card>
   372→
   373→      {/* Table */}
   374→      <Card className="border-slate-200">
   375→        <CardHeader className="pb-4">
   376→          <div className="flex items-center justify-between">
   377→            <div>
   378→              <CardTitle className="text-lg font-semibold text-slate-900">
   379→                Liste des demandes
   380→              </CardTitle>
   381→              <CardDescription>
   382→                {pagination.total} demande(s) trouvée(s)
   383→              </CardDescription>
   384→            </div>
   385→          </div>
   386→        </CardHeader>
   387→        <CardContent>
   388→          {loading ? (
   389→            <div className="space-y-3">
   390→              {[...Array(5)].map((_, i) => (
   391→                <Skeleton key={i} className="h-20 w-full" />
   392→              ))}
   393→            </div>
   394→          ) : requests.length > 0 ? (
   395→            <>
   396→              {/* Desktop Table */}
   397→              <div className="hidden md:block overflow-x-auto">
   398→                <table className="w-full">
   399→                  <thead>
   400→                    <tr className="border-b border-slate-200">
   401→                      <th className="text-left py-3 px-4 text-xs font-semibold text-slate-500 uppercase tracking-wider">Date</th>
   402→                      <th className="text-left py-3 px-4 text-xs font-semibold text-slate-500 uppercase tracking-wider">Demandeur</th>
   403→                      <th className="text-left py-3 px-4 text-xs font-semibold text-slate-500 uppercase tracking-wider">Organisation</th>
   404→                      <th className="text-left py-3 px-4 text-xs font-semibold text-slate-500 uppercase tracking-wider">Type</th>
   405→                      <th className="text-left py-3 px-4 text-xs font-semibold text-slate-500 uppercase tracking-wider">Région</th>
   406→                      <th className="text-left py-3 px-4 text-xs font-semibold text-slate-500 uppercase tracking-wider">Statut</th>
   407→                      <th className="text-right py-3 px-4 text-xs font-semibold text-slate-500 uppercase tracking-wider">Actions</th>
   408→                    </tr>
   409→                  </thead>
   410→                  <tbody>
   411→                    {requests.map((request, index) => {
   412→                      const status = statusConfig[request.statut]
   413→                      return (
   414→                        <tr 
   415→                          key={request.id} 
   416→                          className={`border-b border-slate-100 hover:bg-slate-50 transition-colors ${
   417→                            index % 2 === 0 ? 'bg-white' : 'bg-slate-50/30'
   418→                          }`}
   419→                        >
   420→                          <td className="py-4 px-4">
   421→                            <div className="flex items-center gap-2 text-sm text-slate-600">
   422→                              <Calendar className="w-4 h-4 text-slate-400" />
   423→                              <span>{formatDate(request.created_at)}</span>
   424→                            </div>
   425→                            <span className="text-xs text-slate-400 ml-6">{getRelativeTime(request.created_at)}</span>
   426→                          </td>
   427→                          <td className="py-4 px-4">
   428→                            <div>
   429→                              <p className="font-medium text-slate-900 flex items-center gap-2">
   430→                                <User className="w-4 h-4 text-slate-400" />
   431→                                {request.nom_complet}
   432→                              </p>
   433→                              <p className="text-xs text-slate-500 mt-0.5 flex items-center gap-1">
   434→                                <Mail className="w-3 h-3" />
   435→                                {request.email}
   436→                              </p>
   437→                            </div>
   438→                          </td>
   439→                          <td className="py-4 px-4">
   440→                            <p className="font-medium text-slate-900 flex items-center gap-2">
   441→                              <Building2 className="w-4 h-4 text-slate-400" />
   442→                              {request.nom_organisation}
   443→                            </p>
   444→                          </td>
   445→                          <td className="py-4 px-4">
   446→                            <Badge variant="outline" className="font-mono text-xs bg-orange-50 text-orange-700 border-orange-200">
   447→                              {request.type_org}
   448→                            </Badge>
   449→                          </td>
   450→                          <td className="py-4 px-4">
   451→                            <span className="text-sm text-slate-600 flex items-center gap-1">
   452→                              <MapPin className="w-4 h-4 text-slate-400" />
   453→                              {request.region}
   454→                            </span>
   455→                          </td>
   456→                          <td className="py-4 px-4">
   457→                            <Badge variant="outline" className={`${status.className} text-xs`}>
   458→                              {status.icon && <status.icon className="w-3 h-3 mr-1" />}
   459→                              {status.label}
   460→                            </Badge>
   461→                          </td>
   462→                          <td className="py-4 px-4">
   463→                            <div className="flex items-center justify-end gap-1">
   464→                              {/* View details */}
   465→                              <Dialog open={dialogOpen && selectedRequest?.id === request.id} onOpenChange={(open) => {
   466→                                setDialogOpen(open)
   467→                                if (open) {
   468→                                  setSelectedRequest(request)
   469→                                  setNotesValue(request.notes_admin || '')
   470→                                }
   471→                              }}>
   472→                                <DialogTrigger asChild>
   473→                                  <Button variant="ghost" size="icon" className="h-8 w-8">
   474→                                    <Eye className="w-4 h-4 text-slate-400 hover:text-orange-600" />
   475→                                  </Button>
   476→                                </DialogTrigger>
   477→                                <DialogContent className="max-w-2xl max-h-[90vh] overflow-y-auto">
   478→                                  <DialogHeader>
   479→                                    <DialogTitle>Détails de la demande</DialogTitle>
   480→                                    <DialogDescription>
   481→                                      Demande #{request.id.slice(-8)}
   482→                                    </DialogDescription>
   483→                                  </DialogHeader>
   484→                                  
   485→                                  {selectedRequest && (
   486→                                    <div className="space-y-6 mt-4">
   487→                                      {/* Status badge */}
   488→                                      <div className="flex items-center justify-between p-4 rounded-lg bg-slate-50">
   489→                                        <span className="text-sm text-slate-600">Statut actuel</span>
   490→                                        <Badge variant="outline" className={statusConfig[selectedRequest.statut].className}>
   491→                                          {statusConfig[selectedRequest.statut].label}
   492→                                        </Badge>
   493→                                      </div>
   494→
   495→                                      {/* Info grid */}
   496→                                      <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
   497→                                        <div className="space-y-3">
   498→                                          <div className="flex items-start gap-3">
   499→                                            <User className="w-5 h-5 text-slate-400 mt-0.5" />
   500→                                            <div>
   501→                                              <p className="text-xs text-slate-500">Demandeur</p>
   502→                                              <p className="font-medium text-slate-900">{selectedRequest.nom_complet}</p>
   503→                                            </div>
   504→                                          </div>
   505→                                          <div className="flex items-start gap-3">
   506→                                            <Mail className="w-5 h-5 text-slate-400 mt-0.5" />
   507→                                            <div>
   508→                                              <p className="text-xs text-slate-500">Email</p>
   509→                                              <p className="font-medium text-slate-900">{selectedRequest.email}</p>
   510→                                            </div>
   511→                                          </div>
   512→                                          <div className="flex items-start gap-3">
   513→                                            <Phone className="w-5 h-5 text-slate-400 mt-0.5" />
   514→                                            <div>
   515→                                              <p className="text-xs text-slate-500">Téléphone</p>
   516→                                              <p className="font-medium text-slate-900">{selectedRequest.telephone}</p>
   517→                                            </div>
   518→                                          </div>
   519→                                          {selectedRequest.whatsapp && (
   520→                                            <div className="flex items-start gap-3">
   521→                                              <Phone className="w-5 h-5 text-green-500 mt-0.5" />
   522→                                              <div>
   523→                                                <p className="text-xs text-slate-500">WhatsApp</p>
   524→                                                <p className="font-medium text-slate-900">{selectedRequest.whatsapp}</p>
   525→                                              </div>
   526→                                            </div>
   527→                                          )}
   528→                                          {selectedRequest.fonction && (
   529→                                            <div className="flex items-start gap-3">
   530→                                              <ClipboardList className="w-5 h-5 text-slate-400 mt-0.5" />
   531→                                              <div>
