   111→-- Représente une organisation cliente (DR ou DD du MCNSLP)
   112→-- ----------------------------------------------------------------------------
   113→CREATE TABLE IF NOT EXISTS organizations (
   114→  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   115→  created_at TIMESTAMPTZ DEFAULT NOW(),
   116→  updated_at TIMESTAMPTZ,
   117→  
   118→  -- Informations identitaires
   119→  nom TEXT NOT NULL,                          -- Nom de la direction régionale/départementale
   120→  type_org TEXT NOT NULL CHECK (type_org IN ('DR', 'DD')), -- Direction Régionale ou Départementale
   121→  region TEXT NOT NULL,                        -- Région administrative
   122→  departement TEXT,                           -- Département (pour DD)
   123→  
   124→  -- Coordonnées
   125→  telephone TEXT,                             -- Téléphone principal
   126→  email TEXT,                                 -- Email professionnel
   127→  
   128→  -- Branding
   129→  logo_url TEXT,                              -- URL du logo de l'organisation
   130→  
   131→  -- État
   132→  actif BOOLEAN DEFAULT TRUE,                 -- Organisation active ou désactivée
   133→  
   134→  -- Traçabilité

   459→-- ----------------------------------------------------------------------------
   460→-- Table : activities
   461→-- Activités réalisées (table centrale du système)
   462→-- ----------------------------------------------------------------------------
   463→CREATE TABLE IF NOT EXISTS activities (
   464→  id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
   465→  created_at TIMESTAMPTZ DEFAULT NOW(),
   466→  updated_at TIMESTAMPTZ,
   467→  
   468→  -- Organisation et référence
   469→  organization_id UUID NOT NULL REFERENCES organizations(id) ON DELETE CASCADE,
   470→  reference TEXT UNIQUE NOT NULL,             -- Référence unique auto-générée
   471→  
   472→  -- Dates (avec colonnes générées pour agrégation)
   473→  date_activite DATE NOT NULL,                -- Date de réalisation
   474→  mois INTEGER GENERATED ALWAYS AS (EXTRACT(MONTH FROM date_activite)::INT) STORED,
   475→  trimestre INTEGER GENERATED ALWAYS AS (CEIL(EXTRACT(MONTH FROM date_activite)/3.0)::INT) STORED,
   476→  annee INTEGER GENERATED ALWAYS AS (EXTRACT(YEAR FROM date_activite)::INT) STORED,
   477→  
   478→  -- Localisation
   479→  region TEXT NOT NULL,
   480→  departement TEXT,
   481→  sous_prefecture TEXT,
   482→  
   483→  -- Classification
   484→  category_id UUID REFERENCES activity_categories(id),
   485→  type_id UUID REFERENCES activity_types(id),
   486→  
   487→  -- Contenu
   488→  intitule TEXT NOT NULL,                      -- Titre/intitulé de l'activité
   489→  description TEXT,                            -- Description détaillée
   490→  objectif TEXT,                               -- Objectif de l'activité
   491→  structure_responsable TEXT,                  -- Service/structure responsable
   492→  responsables TEXT[],                         -- Liste des responsables (noms)
   493→  
   494→  -- Résultats et analyse
   495→  resultats TEXT,                              -- Résultats obtenus
   496→  difficultes TEXT,                            -- Difficultés rencontrées
   497→  recommandations TEXT,                        -- Recommandations
   498→  perspectives TEXT,                           -- Perspectives
   499→  observations TEXT,                           -- Observations générales
   500→  
   501→  -- Workflow validation
   502→  statut activity_status DEFAULT 'BROUILLON',
   503→  plan_activity_id UUID,                       -- Lien vers activité planifiée
   504→  est_supplementaire BOOLEAN DEFAULT FALSE,    -- Hors plan initial
   505→  
   506→  -- Traçabilité
   507→  cree_par UUID,                               -- Créateur de l'activité
   508→  soumis_le TIMESTAMPTZ,                       -- Date de soumission
   509→  valide_le TIMESTAMPTZ,                       -- Date de validation
   510→  valide_par UUID,                             -- Validateur
   511→  motif_rejet TEXT                             -- Motif de rejet si applicable
   512→);
   513→
   514→COMMENT ON TABLE activities IS 'Table centrale : activités réalisées par les organisations';
   515→COMMENT ON COLUMN activities.reference IS 'Référence unique formatée (ex: ACT-2024-001)';
   516→COMMENT ON COLUMN activities.mois IS 'Colonne générée : numéro du mois (1-12)';
   517→COMMENT ON COLUMN activities.trimestre IS 'Colonne générée : numéro du trimestre (1-4)';
   518→COMMENT ON COLUMN activities.annee IS 'Colonne générée : année';
   519→COMMENT ON COLUMN activities.est_supplementaire TRUE IS 'Activité non prévue au plan annuel';

   963→-- Calcule le statut effectif d'un abonnement en fonction des dates
   964→-- ----------------------------------------------------------------------------
   965→CREATE OR REPLACE FUNCTION compute_subscription_status(subs subscriptions)
   966→RETURNS subscription_status
   967→LANGUAGE sql
   968→STABLE
   969→AS $$
   970→  CASE
   971→    WHEN subs.statut IN ('RESILIE', 'SUSPENDU') THEN subs.statut
   972→    WHEN subs.date_fin <= NOW() THEN 'EXPIRE'
   973→    WHEN subs.date_fin <= NOW() + INTERVAL '30 days' THEN 'EXPIRANT_BIENTOT'
   974→    ELSE subs.statut
   975→  END;
   976→$$;
   977→

  1035→BEGIN
  1036→  -- Récupération infos utilisateur
  1037→  v_user_id := COALESCE(auth.uid(), '00000000-0000-0000-0000-000000000000'::uuid);
  1038→  
  1039→  -- Tentative récupération organisation (peut échouer si pas de profil)
  1040→  BEGIN
  1041→    SELECT organization_id INTO v_org_id FROM public.profiles WHERE id = v_user_id;
  1042→  EXCEPTION WHEN OTHERS THEN
  1043→    v_org_id := NULL;
  1044→  END;
  1045→  
  1046→  -- Construction des snapshots
  1047→  IF TG_OP = 'DELETE' THEN
  1048→    v_old_json := to_jsonb(OLD);
  1049→    v_new_json := NULL;
  1050→  ELSIF TG_OP = 'INSERT' THEN
  1051→    v_old_json := NULL;
  1052→    v_new_json := to_jsonb(NEW);
  1053→  ELSIF TG_OP = 'UPDATE' THEN
  1054→    v_old_json := to_jsonb(OLD);
  1055→    v_new_json := to_jsonb(NEW);
  1056→  END IF;
  1057→  
  1058→  -- Insertion dans le journal d'audit
  1059→  INSERT INTO public.audit_logs (
  1060→    organization_id,
  1061→    user_id,
  1062→    action,
  1063→    objet_type,
  1064→    objet_id,
  1065→    ancienne_valeur,
  1066→    nouvelle_valeur
  1067→  ) VALUES (
  1068→    v_org_id,
  1069→    v_user_id,
  1070→    TG_OP,
  1071→    TG_TABLE_NAME,
  1072→    COALESCE(NEW.id, OLD.id),
  1073→    v_old_json,
  1074→    v_new_json
  1075→  );
  1076→  
  1077→  -- Pour UPDATE/DELETE, on retourne la ligne originale/modifiée
  1078→  IF TG_OP = 'DELETE' THEN
  1079→    RETURN OLD;
  1080→  ELSE
  1081→    RETURN NEW;
  1082→  END IF;
  1083→END;
  1084→$$;
  1085→
  1086→COMMENT ON FUNCTION audit_trigger_function() IS 'Trigger d''audit : enregistre toutes les modifications dans audit_logs';
