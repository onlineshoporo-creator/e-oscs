   107→export default function OrganisationsPage() {
   108→  const [organizations, setOrganizations] = useState<Organization[]>([])
   109→  const [loading, setLoading] = useState(true)
   110→  const [viewMode, setViewMode] = useState<ViewMode>('grid')
   111→  const [filters, setFilters] = useState<Filters>({
   112→    search: '',
   113→    type: '',
   114→    region: '',
   115→    status: ''
   116→  })
   117→  const [pagination, setPagination] = useState<Pagination>({
   118→    page: 1,
   119→    limit: 12,
   120→    total: 0,
   121→    totalPages: 0
   122→  })
   123→  const [selectedOrg, setSelectedOrg] = useState<Organization | null>(null)
   124→  const [dialogOpen, setDialogOpen] = useState(false)
   125→  const [actionLoading, setActionLoading] = useState(false)
   126→  const [stats, setStats] = useState({ total: 0, actives: 0, dr: 0, dd: 0 })
   127→
   128→  // Fetch organizations
   129→  const fetchOrganizations = useCallback(async () => {
   130→    setLoading(true)
   131→    try {
   132→      const params = new URLSearchParams({
   133→        page: pagination.page.toString(),
   134→        limit: pagination.limit.toString(),
   135→        ...(filters.search && { search: filters.search }),
   136→        ...(filters.type && { type_org: filters.type }),
   137→        ...(filters.region && filters.region !== 'Toutes régions' && { region: filters.region }),
   138→        ...(filters.status !== '' && { actif: filters.status === 'actif' ? 'true' : 'false' })
   139→      })
   140→
   141→      const response = await fetch(`/api/admin/organisations?${params}`)
   142→      if (response.ok) {
   143→        const data = await response.json()
   144→        const orgs = data.data || []
   145→        setOrganizations(orgs)
   146→        setPagination(prev => ({
   147→          ...prev,
   148→          total: data.total || 0,
   149→          totalPages: Math.ceil((data.total || 0) / prev.limit)

   155→          actives: orgs.filter((o: Organization) => o.actif).length,
   156→          dr: orgs.filter((o: Organization) => o.type_org === 'DR').length,
   157→          dd: orgs.filter((o: Organization) => o.type_org === 'DD').length

   160→        toast.error('Erreur lors du chargement des organisations')
   161→      }

   163→      console.error('Erreur:', error)
   164→      toast.error('Erreur de connexion au serveur')

   166→      setLoading(false)
   167→    }
   168→  }, [pagination.page, pagination.limit, filters])
   169→
   170→  useEffect(() => {
   171→    fetchOrganizations()
   172→  }, [fetchOrganizations])
   173→
   174→  // Toggle organization status - utiliser le bon endpoint

   176→    setActionLoading(true)
   177→    try {
   178→      const response = await fetch(`/api/admin/organisations/${orgId}`, {
   179→        method: 'PATCH',
   180→        headers: { 'Content-Type': 'application/json' },
   181→        body: JSON.stringify({ toggleStatus: true })
   182→      })
   183→
   184→      if (response.ok) {
   185→        toast.success(currentStatus 
   186→          ? 'Organisation suspendue avec succès' 
   187→          : 'Organisation réactivée avec succès'
   188→        )
   189→        fetchOrganizations()
   190→        if (selectedOrg?.id === orgId) {
   191→          setSelectedOrg(prev => prev ? { ...prev, actif: !currentStatus } : null)
   192→        }
   193→      } else {
   194→        throw new Error('Erreur lors de la mise à jour')
   195→      }
   196→    } catch (error) {
   197→      console.error(error)
   198→      toast.error('Erreur lors de la modification du statut')
   199→    } finally {
   200→      setActionLoading(false)
   201→    }

   206→      day: 'numeric',
   207→      month: 'short',
   208→      year: 'numeric'
   209→    })
   210→  }
   211→

   213→    if (!org.subscription) return null

   217→    const daysLeft = Math.ceil((endDate.getTime() - now.getTime()) / (1000 * 60 * 60 * 24))
   218→    
   219→    if (daysLeft <= 0) return { label: 'Expiré', className: 'bg-red-100 text-red-700 border-red-200' }
   220→    if (daysLeft <= 7) return { label: `Expire ${daysLeft}j`, className: 'bg-amber-100 text-amber-700 border-amber-200' }

   222→  }
   223→

   225→    <div className="space-y-6">

   227→      <div className="flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">
   228→        <div>
   229→          <h1 className="text-2xl lg:text-3xl font-bold text-slate-900">Organisations</h1>

   240→      {/* Stats cards */}
   241→      <div className="grid grid-cols-2 md:grid-cols-4 gap-4">

   244→            <div className="p-3 rounded-xl bg-blue-100 text-blue-600">
   245→              <Building2 className="w-6 h-6" />
   246→            </div>
   247→            <div>
   248→              <p className="text-2xl font-bold text-slate-900">{stats.total}</p>
   249→              <p className="text-xs text-slate-500">Total organisations</p>
   250→            </div>
   251→          </CardContent>

   256→              <Power className="w-6 h-6" />

   259→              <p className="text-2xl font-bold text-slate-900">{stats.actives}</p>
   260→              <p className="text-xs text-slate-500">Actives</p>
   261→            </div>
   262→          </CardContent>
   263→        </Card>
   264→        <Card className="border-slate-200">
   265→          <CardContent className="p-4 flex items-center gap-4">
   266→            <div className="p-3 rounded-xl bg-orange-100 text-orange-600">
   267→              <Building2 className="w-6 h-6" />
   268→            </div>

   270→              <p className="text-2xl font-bold text-slate-900">{stats.dr}</p>

   311→                <SelectItem value="DD">Direction Départementale</SelectItem>
   312→              </SelectContent>
   313→            </Select>
   314→
   315→            {/* Region filter */}
   316→            <Select value={filters.region} onValueChange={(value) => setFilters(prev => ({ ...prev, region: value }))}>
   317→              <SelectTrigger className="w-full md:w-[180px]">
   318→                <MapPin className="w-4 h-4 mr-2 text-slate-400" />
   319→                <SelectValue placeholder="Région" />
   320→              </SelectTrigger>
   321→              <SelectContent>
   322→                {regionsCI.map(region => (
   323→                  <SelectItem key={region} value={region}>{region}</SelectItem>
   324→                ))}
   325→              </SelectContent>
   326→            </Select>
   327→
   328→            {/* Status filter */}
   329→            <Select value={filters.status} onValueChange={(value) => setFilters(prev => ({ ...prev, status: value }))}>
   330→              <SelectTrigger className="w-full md:w-[140px]">
   331→                <SelectValue placeholder="Statut" />
   332→              </SelectTrigger>
   333→              <SelectContent>
   334→                <SelectItem value="">Tous statuts</SelectItem>
   335→                <SelectItem value="actif">Actives</SelectItem>
   336→                <SelectItem value="inactif">Inactives</SelectItem>
   337→              </SelectContent>
   338→            </Select>
   339→
   340→            {/* View mode toggle */}
   341→            <div className="hidden md:flex items-center border border-slate-200 rounded-lg p-1">
   342→              <Button

   344→                size="icon"
   345→                className={`h-8 w-8 ${viewMode === 'grid' ? 'bg-orange-600 hover:bg-orange-700' : ''}`}
   346→                onClick={() => setViewMode('grid')}
   347→              >
   348→                <Grid3X3 className="w-4 h-4" />
   349→              </Button>
   350→              <Button
   351→                variant={viewMode === 'list' ? 'default' : 'ghost'}
   352→                size="icon"
   353→                className={`h-8 w-8 ${viewMode === 'list' ? 'bg-orange-600 hover:bg-orange-700' : ''}`}
   354→                onClick={() => setViewMode('list')}
   355→              >
   356→                <List className="w-4 h-4" />
   357→              </Button>
   358→            </div>
   359→          </div>
   360→        </CardContent>
   361→      </Card>
   362→
   363→      {/* Content */}
   364→      {loading ? (
   365→        viewMode === 'grid' ? (
   366→          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   367→            {[...Array(6)].map((_, i) => (
   368→              <Card key={i} className="border-slate-200">
   369→                <CardContent className="p-6">
   370→                  <Skeleton className="h-6 w-3/4 mb-4" />
   371→                  <Skeleton className="h-4 w-1/2 mb-2" />
   372→                  <Skeleton className="h-4 w-full mb-4" />
   373→                  <div className="flex justify-between">
   374→                    <Skeleton className="h-8 w-20" />
   375→                    <Skeleton className="h-8 w-20" />
   376→                  </div>
   377→                </CardContent>
   378→              </Card>
   379→            ))}
   380→          </div>
   381→        ) : (
   382→          <div className="space-y-3">
   383→            {[...Array(5)].map((_, i) => (
   384→              <Skeleton key={i} className="h-20 w-full" />
   385→            ))}
   386→          </div>
   387→        )
   388→      ) : organizations.length > 0 ? (
   389→        <>
   390→          {viewMode === 'grid' ? (
   391→            /* Grid View */
   392→            <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-4">
   393→              {organizations.map((org) => (
   394→                <Card key={org.id} className={`border-slate-200 hover:shadow-lg transition-all duration-300 group ${!org.actif ? 'opacity-60' : ''}`}>
   395→                  <CardContent className="p-6">
   396→                    <div className="flex items-start justify-between mb-4">
   397→                      <div className="flex items-center gap-3">
   398→                        <Avatar className="h-12 w-12 ring-2 ring-offset-2 ring-orange-100">
   399→                          <AvatarFallback className={`${
   400→                            org.type_org === 'DR' 
   401→                              ? 'bg-gradient-to-br from-blue-500 to-blue-600' 
   402→                              : 'bg-gradient-to-br from-purple-500 to-purple-600'
   403→                          } text-white font-bold`}>
   404→                            {org.nom.slice(0, 2).toUpperCase()}
   405→                          </AvatarFallback>
   406→                        </Avatar>
   407→                        <div>
   408→                          <h3 className="font-semibold text-slate-900 group-hover:text-orange-600 transition-colors line-clamp-1">
   409→                            {org.nom}
   410→                          </h3>
   411→                          <Badge variant="outline" className={`text-[10px] mt-1 ${
   412→                            org.type_org === 'DR' 
   413→                              ? 'border-blue-200 text-blue-700 bg-blue-50' 
   414→                              : 'border-purple-200 text-purple-700 bg-purple-50'
   415→                          }`}>
   416→                            {org.type_org}
   417→                          </Badge>
   418→                        </div>
   419→                      </div>
   420→                      
   421→                      <DropdownMenu>
   422→                        <DropdownMenuTrigger asChild>
   423→                          <Button variant="ghost" size="icon" className="h-8 w-8 opacity-0 group-hover:opacity-100 transition-opacity">
   424→                            <MoreHorizontal className="w-4 h-4 text-slate-400" />
   425→                          </Button>
   426→                        </DropdownMenuTrigger>
   427→                        <DropdownMenuContent align="end">
   428→                          <DropdownMenuItem onClick={() => { setSelectedOrg(org); setDialogOpen(true); }}>
   429→                            <Eye className="mr-2 h-4 w-4" />
   430→                            Voir détails
   431→                          </DropdownMenuItem>
   432→                          <DropdownMenuSeparator />
   433→                          <DropdownMenuItem 
   434→                            onClick={() => toggleOrgStatus(org.id, org.actif)}
   435→                            disabled={actionLoading}
   436→                            className={org.actif ? 'text-amber-600 focus:text-amber-600' : 'text-green-600 focus:text-green-600'}
   437→                          >
   438→                            {org.actif ? (
   439→                              <>
   440→                                <PowerOff className="mr-2 h-4 w-4" />
   441→                                Suspendre
   442→                              </>
   443→                            ) : (
   444→                              <>
   445→                                <Power className="mr-2 h-4 w-4" />
   446→                                Réactiver
   447→                              </>
   448→                            )}
   449→                          </DropdownMenuItem>
   450→                        </DropdownMenuContent>
   451→                      </DropdownMenu>
   452→                    </div>
   453→
   454→                    <div className="space-y-2 mb-4">
   455→                      <div className="flex items-center gap-2 text-sm text-slate-600">
   456→                        <MapPin className="w-4 h-4 text-slate-400" />
   457→                        <span>{[org.departement, org.region].filter(Boolean).join(', ') || 'Non spécifié'}</span>
   458→                      </div>
   459→                      {org.email && (
   460→                        <div className="flex items-center gap-2 text-sm text-slate-600">
   461→                          <Mail className="w-4 h-4 text-slate-400" />
   462→                          <span className="truncate">{org.email}</span>
   463→                        </div>
   464→                      )}
   465→                      {org.telephone && (
   466→                        <div className="flex items-center gap-2 text-sm text-slate-600">
   467→                          <Phone className="w-4 h-4 text-slate-400" />
   468→                          <span>{org.telephone}</span>
   469→                        </div>
   470→                      )}
   471→                    </div>
   472→
   473→                    <div className="flex items-center justify-between pt-4 border-t border-slate-100">
   474→                      <div className="flex items-center gap-2">
   475→                        <span className={`inline-flex items-center gap-1.5 px-2 py-1 rounded-full text-xs font-medium ${
   476→                          org.actif

   479→                        }`}>
   480→                          <span className={`w-1.5 h-1.5 rounded-full ${org.actif ? 'bg-green-500' : 'bg-red-500'}`} />
   481→                          {org.actif ? 'Active' : 'Inactive'}
   482→                        </span>
   483→                      </div>
   484→                      
   485→                      {(() => {
   486→                        const subStatus = getSubscriptionStatus(org)
   487→                        return subStatus ? (
   488→                          <Badge variant="outline" className={`text-[10px] ${subStatus.className}`}>
   489→                            {subStatus.label}
   490→                          </Badge>
   491→                        ) : null
   492→                      })()}
   493→                    </div>
   494→                  </CardContent>
   495→                </Card>
   496→              ))}
   497→            </div>
   498→          ) : (
   499→            /* List View */
   500→            <Card className="border-slate-200">
   501→              <CardContent className="p-0">

   504→                    <tr className="border-b border-slate-200">
