   966→   966→RETURNS subscription_status

   975→   975→  END;
   976→   976→$$;

  1022→  1022→-- Fonction : audit_trigger_function()
  1023→  1023→-- Trigger générique d'audit qui logue les modifications dans audit_logs
  1024→  1024→-- ----------------------------------------------------------------------------
  1025→  1025→CREATE OR REPLACE FUNCTION audit_trigger_function()
  1026→  1026→RETURNS TRIGGER
  1027→  1027→LANGUAGE plpgsql
  1028→  1028→SECURITY DEFINER
  1029→  1029→AS $$
  1030→  1030→DECLARE
  1031→  1031→  v_old_json JSONB;
  1032→  1032→  v_new_json JSONB;
  1033→  1033→  v_user_id UUID;
  1034→  1034→  v_org_id UUID;
  1035→  1035→BEGIN
  1036→  1036→  -- Récupération infos utilisateur
  1037→  1037→  v_user_id := COALESCE(auth.uid(), '00000000-0000-0000-0000-000000000000'::uuid);
  1038→  1038→  
  1039→  1039→  -- Tentative récupération organisation (peut échouer si pas de profil)
  1040→  1040→  BEGIN
  1041→  1041→    SELECT organization_id INTO v_org_id FROM public.profiles WHERE id = v_user_id;
  1042→  1042→  EXCEPTION WHEN OTHERS THEN
  1043→  1043→    v_org_id := NULL;
  1044→  1044→  END;
  1045→  1045→  
  1046→  1046→  -- Construction des snapshots
  1047→  1047→  IF TG_OP = 'DELETE' THEN
  1048→  1048→    v_old_json := to_jsonb(OLD);
  1049→  1049→    v_new_json := NULL;
  1050→  1050→  ELSIF TG_OP = 'INSERT' THEN
  1051→  1051→    v_old_json := NULL;
  1052→  1052→    v_new_json := to_jsonb(NEW);
  1053→  1053→  ELSIF TG_OP = 'UPDATE' THEN
  1054→  1054→    v_old_json := to_jsonb(OLD);
  1055→  1055→    v_new_json := to_jsonb(NEW);
  1056→  1056→  END IF;
  1057→  1057→  
  1058→  1058→  -- Insertion dans le journal d'audit
  1059→  1059→  INSERT INTO public.audit_logs (
  1060→  1060→    organization_id,
  1061→  1061→    user_id,
  1062→  1062→    action,
  1063→  1063→    objet_type,
  1064→  1064→    objet_id,
  1065→  1065→    ancienne_valeur,
  1066→  1066→    nouvelle_valeur
  1067→  1067→  ) VALUES (
  1068→  1068→    v_org_id,
  1069→  1069→    v_user_id,
  1070→  1070→    TG_OP,
  1071→  1071→    TG_TABLE_NAME,
  1072→  1072→    COALESCE(NEW.id, OLD.id),
  1073→  1073→    v_old_json,
  1074→  1074→    v_new_json
  1075→  1075→  );
  1076→  1076→  
  1077→  1077→  -- Pour UPDATE/DELETE, on retourne la ligne originale/modifiée
  1078→  1078→  IF TG_OP = 'DELETE' THEN
  1079→  1079→    RETURN OLD;
  1080→  1080→  ELSE
  1081→  1081→    RETURN NEW;
  1082→  1082→  END IF;
  1083→  1083→END;
  1084→  1084→$$;
  1085→  1085→
  1086→  1086→COMMENT ON FUNCTION audit_trigger_function() IS 'Trigger d''audit : enregistre toutes les modifications dans audit_logs';
