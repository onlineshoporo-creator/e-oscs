    17→// Hook personnalisé pour animation typing
    18→function useTypingEffect(text: string, speed: number = 50, startDelay: number = 500) {
    19→  const [displayText, setDisplayText] = useState("");
    20→  const [isComplete, setIsComplete] = useState(false);
    21→  const [hasStarted, setHasStarted] = useState(false);
    22→
    23→  useEffect(() => {
    24→    let timeoutId: NodeJS.Timeout;
    25→    
    26→    // Délai avant de commencer
    27→    const startTimeout = setTimeout(() => {
    28→      setHasStarted(true);
    29→    }, startDelay);
    30→
    31→    if (!hasStarted) return;
    32→
    33→    let currentIndex = 0;
    34→    
    35→    const type = () => {
    36→      if (currentIndex <= text.length) {
    37→        setDisplayText(text.slice(0, currentIndex));
    38→        currentIndex++;
    39→        timeoutId = setTimeout(type, speed);
    40→      } else {
    41→        setIsComplete(true);
    42→      }
    43→    };
    44→
    45→    timeoutId = setTimeout(type, speed);
    46→
    47→    return () => {
    48→      clearTimeout(startTimeout);
    49→      clearTimeout(timeoutId);
    50→    };
    51→  }, [text, speed, startDelay, hasStarted]);
    52→
    53→  return { displayText, isComplete };
    54→}

    56→// Composant compteur animé
    57→function AnimatedCounter({ 
    58→  target, 
    59→  suffix = "", 
    60→  prefix = "",
    61→  duration = 2000,
    62→  className = "" 
    63→}: { 
    64→  target: number | string; 
    65→  suffix?: string; 
    66→  prefix?: string;
    67→  duration?: number;
    68→  className?: string;
    69→}) {
    70→  const [count, setCount] = useState(0);
    71→  const [isVisible, setIsVisible] = useState(false);
    72→  const ref = useRef<HTMLSpanElement>(null);
    73→  const numericTarget = typeof target === 'string' ? parseInt(target) || 0 : target;
    74→
    75→  useEffect(() => {
    76→    const observer = new IntersectionObserver(
    77→      ([entry]) => {
    78→        if (entry.isIntersecting && !isVisible) {
    79→          setIsVisible(true);
    80→        }
    81→      },
    82→      { threshold: 0.5 }
    83→    );
    84→
    85→    if (ref.current) {
    86→      observer.observe(ref.current);
    87→    }
    88→
    89→    return () => observer.disconnect();
    90→  }, [isVisible]);
    91→
    92→  useEffect(() => {
    93→    if (!isVisible) return;
    94→
    95→    const prefersReducedMotion = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
    96→    if (prefersReducedMotion) {
    97→      setCount(numericTarget);
    98→      return;
    99→    }
   100→
   101→    let startTime: number;
   102→    const step = (timestamp: number) => {
   103→      if (!startTime) startTime = timestamp;
   104→      const progress = Math.min((timestamp - startTime) / duration, 1);
   105→      
   106→      // Easing function (ease-out-expo)
   107→      const easeOutExpo = progress === 1 ? 1 : 1 - Math.pow(2, -10 * progress);
   108→      setCount(Math.floor(easeOutExpo * numericTarget));
   109→      
   110→      if (progress < 1) {
   111→        requestAnimationFrame(step);
   112→      }
   113→    };
   114→
   115→    requestAnimationFrame(step);
   116→  }, [isVisible, numericTarget, duration]);
   117→
   118→  return (
   119→    <span ref={ref} className={className}>
   120→      {prefix}{count}{suffix}
   121→    </span>
   122→  );
   123→}
