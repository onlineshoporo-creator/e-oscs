    57→  // Étape 2: Description

    78→// Configuration des étapes

    80→  {

    86→  {

   104→]

   107→const typesActivite = [

   113→  { value: 'REUNION', label: 'Réunion', emoji: '👥' },

   115→]
   116→
   117→// Données initiales du formulaire
   118→const initialFormData: ActivityFormData = {
   119→  intitule: '',
   120→  dateActivite: '',
   121→  heureDebut: '',
   122→  heureFin: '',
   123→  lieu: '',
   124→  typeActivite: '',
   125→  objectif: '',
   126→  description: '',
   127→  resultats: '',
   128→  totalBeneficiaires: '',
   129→  hommes: '',
   130→  femmes: '',
   131→  enfants: '',
   132→  personnesHandicapees: '',
   133→  documents: [],
   134→  documentPreviews: [],
   135→}
   136→
   137→export default function NouvelleActivitePage() {
   138→  const router = useRouter()
   139→  const { toast } = useToast()
   140→  
   141→  // États du formulaire
   142→  const [currentStep, setCurrentStep] = useState(1)
   143→  const [formData, setFormData] = useState<ActivityFormData>(initialFormData)
   144→  const [errors, setErrors] = useState<FormErrors>({})
   145→  const [isSubmitting, setIsSubmitting] = useState(false)
   146→  const [showPreview, setShowPreview] = useState(false)
   147→  const [lastSaved, setLastSaved] = useState<Date | null>(null)
   148→  const [isDragging, setIsDragging] = useState(false)
   149→  
   150→  // Calculer la progression globale
   151→  const calculateProgress = (): number => {
   152→    let filledFields = 0
   153→    let totalFields = 0
   154→    
   155→    // Champs obligatoires à vérifier
   156→    const fieldsToCheck = [
   157→      formData.intitule,
   158→      formData.dateActivite,
   159→      formData.lieu,
   160→      formData.typeActivite,
   161→      formData.objectif,
   162→      formData.description,
   163→      formData.totalBeneficiaires,
   164→    ]
   165→    
   166→    totalFields = fieldsToCheck.length
   167→    filledFields = fieldsToCheck.filter(f => f && f.trim().length > 0).length
   168→    
   169→    return Math.round((filledFields / totalFields) * 100)
   170→  }
   171→  
   172→  const progress = calculateProgress()
   173→
   174→  // Auto-save toutes les 30 secondes
   175→  useEffect(() => {
   176→    const interval = setInterval(() => {
   177→      if (formData.intitule || formData.description) {
   178→        saveDraft()
   179→      }
   180→    }, 30000)
   181→    
   182→    return () => clearInterval(interval)
   183→  }, [formData])
   184→
   185→  // Sauvegarder le brouillon
   186→  const saveDraft = useCallback(() => {
   187→    // Simuler la sauvegarde (en production: appel API)
   188→    localStorage.setItem('activity_draft', JSON.stringify({
   189→      ...formData,
   190→      savedAt: new Date().toISOString(),
   191→    }))
   192→    setLastSaved(new Date())
   193→    
   194→    toast({
   195→      title: 'Brouillon sauvegardé',
   196→      description: 'Votre progression a été enregistrée.',
   197→    })
   198→  }, [formData, toast])
   199→
   200→  // Charger un brouillon existant au montage
   201→  useEffect(() => {
   202→    const savedDraft = localStorage.getItem('activity_draft')
   203→    if (savedDraft) {
   204→      try {
   205→        const draft = JSON.parse(savedDraft)
   206→        if (draft.savedAt) {
   207→          setLastSaved(new Date(draft.savedAt))
   208→          toast({
   209→            title: 'Brouillon récupéré',
   210→            description: 'Voulez-vous reprendre votre activité en cours ?',
   211→            action: (
   212→              <Button 
   213→                size="sm" 
   214→                onClick={() => {
   215→                  setFormData({ ...initialFormData, ...draft })
   216→                  toast({ title: 'Brouillon restauré' })
   217→                }}
   218→              >
   219→                Restaurer
   220→              </Button>
   221→            ),
   222→          })
   223→        }
   224→      } catch (e) {
   225→        console.error('Erreur lors du chargement du brouillon')
   226→      }
   227→    }
   228→  }, [])
   229→
   230→  // Validation temps réel
   231→  const validateField = (field: string, value: string): string | null => {
   232→    switch (field) {
   233→      case 'intitule':
   234→        if (!value.trim()) return "L'intitulé est obligatoire"
   235→        if (value.trim().length < 5) return "L'intitulé doit contenir au moins 5 caractères"
   236→        if (value.trim().length > 200) return "L'intitulé ne doit pas dépasser 200 caractères"
   237→        break
   238→      case 'dateActivite':
   239→        if (!value) return "La date est obligatoire"
   240→        const selectedDate = new Date(value)
   241→        const today = new Date()
   242→        today.setHours(0, 0, 0, 0)
   243→        if (selectedDate > today) return "La date ne peut pas être dans le futur"
   244→        break
   245→      case 'lieu':
   246→        if (!value.trim()) return "Le lieu est obligatoire"
   247→        break
   248→      case 'typeActivite':
   249→        if (!value) return "Le type d'activité est obligatoire"
   250→        break
   251→      case 'objectif':
   252→        if (!value.trim()) return "L'objectif est obligatoire"
   253→        break
   254→      case 'description':
   255→        if (!value.trim()) return "La description est obligatoire"
   256→        if (value.trim().length < 20) return "La description doit contenir au moins 20 caractères"
   257→        break
   258→      case 'totalBeneficiaires':
   259→        if (!value) return "Le nombre de bénéficiaires est obligatoire"
   260→        const num = parseInt(value)
   261→        if (isNaN(num) || num < 1) return "Entrez un nombre valide"
   262→        if (num > 10000) return "Le nombre semble trop élevé"
   263→        break
   264→    }
   265→    return null
   266→  }
   267→
   268→  // Gérer les changements avec validation temps réel
   269→  const handleChange = (field: string, value: string) => {
   270→    setFormData(prev => ({ ...prev, [field]: value }))
   271→    
   272→    // Validation temps réel
   273→    const error = validateField(field, value)
   274→    setErrors(prev => {
   275→      if (error) {
   276→        return { ...prev, [field]: error }
   277→      } else {
   278→        const next = { ...prev }
   279→        delete next[field]
   280→        return next
   281→      }
   282→    })
   283→  }
   284→
   285→  // Valider l'étape actuelle
   286→  const validateCurrentStep = (): boolean => {
   287→    const newErrors: FormErrors = {}
   288→    
   289→    switch (currentStep) {
   290→      case 1:
   291→        const step1Fields = ['intitule', 'dateActivite', 'lieu', 'typeActivite']
   292→        step1Fields.forEach(field => {
   293→          const error = validateField(field, (formData as any)[field])
   294→          if (error) newErrors[field] = error
   295→        })
   296→        break
   297→      case 2:
   298→        const step2Fields = ['objectif', 'description']
   299→        step2Fields.forEach(field => {
   300→          const error = validateField(field, (formData as any)[field])
   301→          if (error) newErrors[field] = error
   302→        })
   303→        break
   304→      case 3:
   305→        const error = validateField('totalBeneficiaires', formData.totalBeneficiaires)
   306→        if (error) newErrors.totalBeneficiaires = error
   307→        break
   308→      case 4:
   309→        // Pas de champs obligatoires pour les documents
   310→        break
   311→    }
   312→    
   313→    setErrors(newErrors)
   314→    return Object.keys(newErrors).length === 0
   315→  }
   316→
   317→  // Navigation entre étapes
   318→  const nextStep = () => {
   319→    if (validateCurrentStep()) {
   320→      if (currentStep < steps.length) {
   321→        setCurrentStep(currentStep + 1)
   322→        window.scrollTo({ top: 0, behavior: 'smooth' })
   323→      }
   324→    } else {
   325→      toast({
   326→        title: 'Champs manquants',
   327→        description: 'Veuillez remplir tous les champs obligatoires.',
   328→        variant: 'destructive',
   329→      })
   330→    }
   331→  }
   332→
   333→  const prevStep = () => {
   334→    if (currentStep > 1) {
   335→      setCurrentStep(currentStep - 1)
   336→      window.scrollTo({ top: 0, behavior: 'smooth' })
   337→    }
   338→  }
   339→
   340→  // Gestion des fichiers (drag & drop)
   341→  const handleFileUpload = (files: FileList | null) => {
   342→    if (!files) return
   343→    
   344→    const newFiles = Array.from(files).filter(file => 
   345→      file.type.startsWith('image/') && file.size <= 5 * 1024 * 1024
   346→    )
   347→    
   348→    if (newFiles.length !== files.length) {
   349→      toast({
   350→        title: 'Fichiers ignorés',
   351→        description: 'Seules les images de moins de 5MB sont acceptées.',
   352→        variant: 'destructive',
   353→      })
   354→    }
   355→    
   356→    newFiles.forEach(file => {
   357→      const reader = new FileReader()
   358→      reader.onload = (e) => {
   359→        setFormData(prev => ({
   360→          ...prev,
   361→          documents: [...prev.documents, file],
   362→          documentPreviews: [...prev.documentPreviews, e.target?.result as string],
   363→        }))
   364→      }
   365→      reader.readAsDataURL(file)
   366→    })
   367→  }
   368→
   369→  const removeDocument = (index: number) => {
   370→    setFormData(prev => ({
   371→      ...prev,
   372→      documents: prev.documents.filter((_, i) => i !== index),
   373→      documentPreviews: prev.documentPreviews.filter((_, i) => i !== index),
   374→    }))
   375→  }
   376→
   377→  const handleDragOver = (e: React.DragEvent) => {
   378→    e.preventDefault()
   379→    setIsDragging(true)
   380→  }
   381→
   382→  const handleDragLeave = () => {
   383→    setIsDragging(false)
   384→  }
   385→
   386→  const handleDrop = (e: React.DragEvent) => {
   387→    e.preventDefault()
   388→    setIsDragging(false)
   389→    handleFileUpload(e.dataTransfer.files)
   390→  }
   391→
   392→  // Soumission finale
   393→  const handleSubmit = async () => {
   394→    if (!validateCurrentStep()) return
   395→    
   396→    setIsSubmitting(true)
   397→    
   398→    try {
   399→      // Simuler l'envoi API
   400→      await new Promise(resolve => setTimeout(resolve, 2000))
   401→      
   402→      // Nettoyer le brouillon
   403→      localStorage.removeItem('activity_draft')
   404→      
   405→      toast({
   406→        title: 'Activité soumise avec succès ! 🎉',
   407→        description: 'Votre activité a été envoyée pour validation.',
   408→      })
   409→      
   410→      router.push('/agent/soumises')
   411→    } catch (error) {
   412→      toast({
   413→        title: 'Erreur lors de la soumission',
   414→        description: 'Une erreur est survenue. Veuillez réessayer.',
   415→        variant: 'destructive',
   416→      })
   417→    } finally {
   418→      setIsSubmitting(false)
   419→    }
   420→  }
   421→
   422→  // Rendu de chaque étape
   423→  const renderStepContent = () => {
   424→    switch (currentStep) {
   425→      case 1:
   426→        return (
   427→          <div className="space-y-6">
   428→            <div className="grid grid-cols-1 md:grid-cols-2 gap-6">
   429→              {/* Intitulé */}
   430→              <div className="md:col-span-2 space-y-2">
   431→                <Label htmlFor="intitule" className="text-base font-medium flex items-center gap-2">
   432→                  Intitulé de l&apos;activité <span className="text-red-500">*</span>
   433→                </Label>
   434→                <Input
   435→                  id="intitule"
   436→                  placeholder="Ex: Sensibilisation sur l'hygiène communautaire"
   437→                  value={formData.intitule}
   438→                  onChange={(e) => handleChange('intitule', e.target.value)}
   439→                  className={`h-12 ${errors.intitule ? 'border-red-500 focus-visible:ring-red-500' : ''}`}
   440→                />
   441→                {errors.intitule && (
   442→                  <p className="text-sm text-red-500 flex items-center gap-1">
   443→                    <AlertCircle className="w-4 h-4" />
   444→                    {errors.intitule}
   445→                  </p>
   446→                )}
   447→                <p className="text-xs text-slate-400">{formData.intitule.length}/200 caractères</p>
   448→              </div>
   449→
   450→              {/* Type d'activité */}
   451→              <div className="space-y-2">
   452→                <Label className="text-base font-medium flex items-center gap-2">
   453→                  Type d&apos;activité <span className="text-red-500">*</span>
   454→                </Label>
   455→                <Select value={formData.typeActivite} onValueChange={(v) => handleChange('typeActivite', v)}>
   456→                  <SelectTrigger className={`h-12 ${errors.typeActivite ? 'border-red-500' : ''}`}>
   457→                    <SelectValue placeholder="Sélectionner le type" />
   458→                  </SelectTrigger>
   459→                  <SelectContent>
   460→                    {typesActivite.map(type => (
   461→                      <SelectItem key={type.value} value={type.value}>
   462→                        <span className="flex items-center gap-2">
   463→                          <span>{type.emoji}</span>
   464→                          {type.label}
   465→                        </span>
   466→                      </SelectItem>
   467→                    ))}
   468→                  </SelectContent>
   469→                </Select>
   470→                {errors.typeActivite && (
   471→                  <p className="text-sm text-red-500 flex items-center gap-1">
   472→                    <AlertCircle className="w-4 h-4" />
   473→                    {errors.typeActivite}
   474→                  </p>
   475→                )}
   476→              </div>
   477→
   478→              {/* Date */}
   479→              <div className="space-y-2">
   480→                <Label htmlFor="dateActivite" className="text-base font-medium flex items-center gap-2">
   481→                  Date de l&apos;activité <span className="text-red-500">*</span>
   482→                </Label>
   483→                <Input
   484→                  id="dateActivite"
   485→                  type="date"
   486→                  value={formData.dateActivite}
   487→                  onChange={(e) => handleChange('dateActivite', e.target.value)}
   488→                  className={`h-12 ${errors.dateActivite ? 'border-red-500' : ''}`}
   489→                  max={new Date().toISOString().split('T')[0]}
   490→                />
   491→                {errors.dateActivite && (
   492→                  <p className="text-sm text-red-500 flex items-center gap-1">
   493→                    <AlertCircle className="w-4 h-4" />
   494→                    {errors.dateActivite}
   495→                  </p>
   496→                )}
   497→              </div>
   498→
   499→              {/* Heures */}
   500→              <div className="space-y-2">
   501→                <Label htmlFor="heureDebut" className="text-base font-medium">Heure de début</Label>
   502→                <Input
   503→                  id="heureDebut"
   504→                  type="time"
   505→                  value={formData.heureDebut}
   506→                  onChange={(e) => handleChange('heureDebut', e.target.value)}
   507→                  className="h-12"
   508→                />
   509→              </div>
   510→
   511→              <div className="space-y-2">
   512→                <Label htmlFor="heureFin" className="text-base font-medium">Heure de fin</Label>
   513→                <Input
   514→                  id="heureFin"
   515→                  type="time"
   516→                  value={formData.heureFin}
   517→                  onChange={(e) => handleChange('heureFin', e.target.value)}
   518→                  className="h-12"
   519→                />
   520→              </div>
   521→
   522→              {/* Lieu */}
   523→              <div className="md:col-span-2 space-y-2">
   524→                <Label htmlFor="lieu" className="text-base font-medium flex items-center gap-2">
   525→                  <MapPin className="w-4 h-4 text-[#F77F00]" />
   526→                  Lieu <span className="text-red-500">*</span>
   527→                </Label>
   528→                <Input
   529→                  id="lieu"
   530→                  placeholder="Ex: Abidjan, Yopougon - Salle communautaire"
   531→                  value={formData.lieu}
   532→                  onChange={(e) => handleChange('lieu', e.target.value)}
   533→                  className={`h-12 ${errors.lieu ? 'border-red-500' : ''}`}
   534→                />
   535→                {errors.lieu && (
   536→                  <p className="text-sm text-red-500 flex items-center gap-1">
   537→                    <AlertCircle className="w-4 h-4" />
   538→                    {errors.lieu}
   539→                  </p>
   540→                )}
   541→              </div>
   542→            </div>
   543→          </div>
   544→        )
   545→
   546→      case 2:
   547→        return (
   548→          <div className="space-y-6">
   549→            {/* Objectif */}
   550→            <div className="space-y-2">
   551→              <Label htmlFor="objectif" className="text-base font-medium flex items-center gap-2">
   552→                <Target className="w-4 h-4 text-[#F77F00]" />
   553→                Objectif principal <span className="text-red-500">*</span>
   554→              </Label>
   555→              <Textarea
   556→                id="objectif"
   557→                placeholder="Décrivez l'objectif principal de cette activité..."
   558→                value={formData.objectif}
   559→                onChange={(e) => handleChange('objectif', e.target.value)}
   560→                rows={3}
   561→                className={`resize-none ${errors.objectif ? 'border-red-500 focus-visible:ring-red-500' : ''}`}
   562→              />
   563→              {errors.objectif && (
   564→                <p className="text-sm text-red-500 flex items-center gap-1">
   565→                  <AlertCircle className="w-4 h-4" />
   566→                  {errors.objectif}
   567→                </p>
   568→              )}
   569→            </div>
   570→
   571→            {/* Description */}
   572→            <div className="space-y-2">
   573→              <Label htmlFor="description" className="text-base font-medium flex items-center gap-2">
   574→                <FileText className="w-4 h-4 text-[#F77F00]" />
   575→                Description détaillée <span className="text-red-500">*</span>
   576→              </Label>
   577→              <Textarea
   578→                id="description"
   579→                placeholder="Décrivez en détail le déroulement de l'activité, les sujets abordés, les méthodes utilisées..."
   580→                value={formData.description}
   581→                onChange={(e) => handleChange('description', e.target.value)}
   582→                rows={5}
   583→                className={`resize-none ${errors.description ? 'border-red-500 focus-visible:ring-red-500' : ''}`}
   584→              />
   585→              {errors.description && (
   586→                <p className="text-sm text-red-500 flex items-center gap-1">
   587→                  <AlertCircle className="w-4 h-4" />
   588→                  {errors.description}
   589→                </p>
   590→              )}
   591→              <p className="text-xs text-slate-400">{formData.description.length} caractères (minimum 20)</p>
   592→            </div>
   593→
   594→            {/* Résultats */}
   595→            <div className="space-y-2">
   596→              <Label htmlFor="resultats" className="text-base font-medium flex items-center gap-2">
   597→                <Award className="w-4 h-4 text-[#009E60]" />
   598→                Résultats obtenus
   599→              </Label>
   600→              <Textarea
   601→                id="resultats"
